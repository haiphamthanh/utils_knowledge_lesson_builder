  {\normalfont\sffamily\bfseries}
  {}
  {0pt}
  {%
    \begin{tikzpicture}
      \node[
        fill=BannerBlue,
        text=white,
        align=left,
        text width=.72\textwidth,
        minimum height=34mm,
        inner xsep=9mm,
        inner ysep=7mm,
        font=\fontsize{25}{30}\selectfont\bfseries
      ] (chapterbar) {#1};
      \path[fill=BannerNavy]
        (chapterbar.north west) rectangle ([yshift=-7mm]chapterbar.north east);
      \node[
        signal,
        signal from=north,
        signal pointer angle=150,
        fill=BannerCream,
        drop shadow={shadow xshift=0pt,shadow yshift=-1.5pt},
        rotate=-90,
        minimum width=39mm,
        minimum height=30mm,
        anchor=north
      ] (chaptertab) at ([xshift=18mm,yshift=9mm]chapterbar.north east) {};
      \node[font=\sffamily\bfseries\footnotesize,color=BannerNavy,rotate=90]
        at ([xshift=-8mm]chaptertab.center) {CHƯƠNG};
      \node[font=\sffamily\bfseries\fontsize{35}{35}\selectfont,color=BannerBlue,rotate=90]
        at ([xshift=7mm]chaptertab.center) {\thechapter};
      \fill[BannerGold] (chapterbar.south west) rectangle
        ([yshift=2.2mm]chapterbar.south east);
    \end{tikzpicture}%
  }
